cmake_minimum_required(VERSION 3.16)
project(orbit_pi C)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ORBIT_BUILD_POSIX ON CACHE BOOL "" FORCE)
set(ORBIT_SDK_DIR ${CMAKE_CURRENT_LIST_DIR}/../../../sdk/embedded CACHE PATH "Orbit embedded SDK")
add_subdirectory(${ORBIT_SDK_DIR} orbit)
add_executable(orbit_pi main.c ../common/client.c)
target_include_directories(orbit_pi PRIVATE ../common)
target_link_libraries(orbit_pi PRIVATE orbit_embedded_posix)
set(ORBIT_APP_KEY "" CACHE STRING "Public app key from the Orbit Integration page")
if(ORBIT_APP_KEY)
    target_compile_definitions(orbit_pi PRIVATE ORBIT_APP_KEY=\"${ORBIT_APP_KEY}\")
endif()
